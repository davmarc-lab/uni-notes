\section{Linguaggi di Programmazione}\label{sec:i_linguaggi_di_programmazione} % (fold)
\begin{definition}{Linguaggio di programmazione}{}
    È uno strumento per descrivere in modo rigoroso \textbf{procedure} per la risoluzione di problemi.
\end{definition}
\begin{note}
    Compilatore: front-end (parser, \dots) + code generation
\end{note}
Generazione automatica del codice del parser per un nuovo linguaggio tramite un approccio dichiarativo:
\begin{itemize}
    \item lessico: espressioni regolari (o automi a stati finiti);
    \item sintassi: grammatiche (EBNF o automi a pila);
\end{itemize}
\begin{note}[Parser]
    Funzionamento del parser generato:
    \begin{itemize}
        \item input: sorgente di un programma;
        \item output: elenco errori lessicali/sintattici o, se \textbf{privo di errori}, un \red{albero sintattico} del programma;
    \end{itemize}
\end{note}

\section{Linguaggi Naturali}\label{sec:i_linguaggi_naturali} % (fold)
\begin{definition}{Frasi Ambigue}{}
    Data una frase in input, anche se corretta, il suo albero sintattico può \textbf{non essere unico}
\end{definition}
\begin{definition}{Grammatiche Probabilistiche}{}
    Data una frase si può inferire l'\textbf{albero sintattico} più probabile
\end{definition}
%
\subsection{Language Model}\label{sub:i_language_model} % (fold)
Oltre a \textbf{grammatiche probabilistiche}, altri esempi classici di \ac{lm}:
\begin{itemize}
    \item \textbf{Bag of Words Model};
    \item \textbf{Hidden Markov Model} cioè automi a stati finiti probabilistici; 
\end{itemize}
Più recentemente \ac{lm} basati su \textbf{modelli neurali}:
\begin{itemize}
    \item nell'approccio classico basato su modelli probabilistici molta conoscenza linguistica viene \red{esplicitamente} rappresentata;
    \item nei modelli neurali le regolarità lessicali e sintattiche emergono \textbf{implicitamente} dei dati nell'apprendimento;
\end{itemize}
%
\subsubsection{Modelli Computazionali}\label{sub:i_modelli_computazionali} % (fold)
In un \ac{lm} è possibile fare \textbf{computazioni} tramite algoritmi:
\begin{itemize}
    \item \textbf{supervised/unsupervised learning};
    \item \textbf{inference/classification};
    \item \textbf{data generation} tramite campionamento distribuzioni;
\end{itemize}
In questo senso gli \ac{lm}, come tutti i \textbf{Generative Model}, sono \textbf{modelli computazionali}.
\emptyln{}
Ma anche i \textbf{Discriminative Model}, alternativi ai Generative Model, sono modelli computazionali specializzati in \textbf{classification}, \textbf{logic regression} e \textbf{Convolutional Neural Networks}.

\subsubsection{Logiche di base ed Estensioni}\label{sub:i_logiche_di_base_ed_estensioni} % (fold)
\begin{description}
    \item[Linear Time Logic] Sono utili per:
        \begin{itemize}
            \item rappresentare la \textbf{conoscenza negli agenti intelligenti} (fondamenti matematici);
            \item rappresentare la \textbf{semantica dei linguaggi naturali};
            \item rappresentare e verificare (\textit{model checking}) proprietà di \textbf{sistemi distribuiti};
        \end{itemize}
\end{description}

\subsubsection{Sistemi Distribuiti e Model Checking}\label{sub:i_sistemi_distribuiti_e_model_checking} % (fold)
Per rappresentare i sistemi \textbf{distribuiti/concorrenti}, ovvero multipli programmi comunicanti tra loro:
\begin{itemize}
    \item \textbf{model checking}: stabilisce algoritmicamente se un sistema \textbf{soddisfa} o meno una data \textbf{proprietà} di logica temporale;
    \item \textbf{modelli computazionali}: \ac{lts}, \textbf{Process Algebra} e \textbf{Petri Net};
\end{itemize}

\begin{note}[Logiche Temporali]
    Permettono di esprimere le \textbf{proprietà} del tipo: \textit{non è possibile raggiungere uno stato in cui:}
    \begin{itemize}
        \item \textit{il negozio ha spedito;}
        \item \textit{non è possibile raggiungere il trasferimento del denaro;}
    \end{itemize}
    \begin{definition}{Model Checker}{}
        Strumenti che verificano se \textbf{proprietà} espresse tramite \textbf{logiche temporali} valgono oppure no.
    \end{definition}
\end{note}
