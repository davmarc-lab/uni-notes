\section{Concetti di Base}\label{sec:lr_concetti_di_base} % (fold)
\begin{definition}{Alfabeto}{alfabeto}
    Insieme finito e non vuoto di simboli.
    \begin{itemize}
        \item \textbf{binario}: $\Si = \pg{0,1}$
        \item lettere: $\Si = \pg{a,b,c,\dots,z}$
        \item insieme di tutti i caratteri ASCII
    \end{itemize}
\end{definition}
\begin{definition}{Stringa}{}
    Sequenza finita di simboli da un \textbf{alfabeto} $\Si$.
    \begin{example}{}{}
        $0011001$ è una stringa su un alfabeto $\pg{0,1}$
    \end{example}
\end{definition}
\begin{definition}{Singoletto}{}
    Un insieme di un elemento si definisce \textbf{singoletto}.
\end{definition}
\begin{definition}{Stringa Vuota}{}
    La \textbf{stringa} con zero occorrenze di simboli da $\Si$.
    \begin{note}
        La stringa vuota è denotata con $\eps$
    \end{note}
\end{definition}
\begin{definition}{Linghezza di una stringa}{}
    Numero di posizioni per i simboli nella stringa.
    $\ab{w}$ denota la lunghezza della stringa $w$
\end{definition}
\begin{definition}{Potenze}{}
    La \textbf{potenza} di un alfabeto si denota con $\Si[k]$ e corrisponde all'insieme delle stringhe di lunghezza $k$ con simboli da $\Si$
    \begin{itemize}
        \item $\Si = \pg{0,1}$
        \item $\Si[1] = \pg{0,1}$
        \item $\Si[2] = \pg{00, 01, 10, 11}$
        \item $\Si[0] = \pg{\eps}$
    \end{itemize}
\end{definition}
\begin{note}
    L'inseme di tutte le stringhe su $\Si$ è denotato da $\Si[*]$
    \[
        \Si[+] = \Si[1] \cup\Si[2] \cup  \Si[3] \cup\dots
    \]
    \[
        \Si[*] = \Si[+] \cup \pg \eps
    \]
\end{note}
\begin{definition}{Concatenzazione}{}
    Se $x$ e $y$ sono stringhe, allora $xy$ è la stringa ottenuta collocando una copia di $y$ subito dopo una copia di $x$.
    \[
        \begin{aligned}
            &x = a_1a_2\,\dots\,a_i \qquad y = b_1b_2\,\dots\,b_j \\
            &xy = a_1a_2\,\dots\,a_ib_1b_2\,\dots\,b_j
        \end{aligned}
    \]
    \begin{note}
        Per ogni stringa $x$:
        \[
            x\eps = \eps x = x
        \]
    \end{note}
\end{definition}
\begin{definition}{Linguaggi formali}{linguaggio}
    Se $\Si$ è un alfabeto, e $L \subseteq \Si[*]$ allora $L$ è un \textbf{linguaggio formale}
    \begin{note}[Linguaggio Vuoto]
        \[
            \emptyset
        \]
        Il linguaggio vuoto è diverso da $\pg{\eps}$
    \end{note}
\end{definition}

\subsection{Automi a Stati Finiti Deterministici}\label{sub:lr_automi_a_stati_finiti_deterministici} % (fold)
Un \ac{dfa} è una \textbf{quintupla}:
\[
    A = \pt{Q,\Si,\dt, q_0, F}
\]
\begin{itemize}
    \item $Q$ è un insieme \textbf{finito di stati};
    \item $\Si$ è un alfabeto finito;
    \item $\dt$ è una funzione di transizione da $Q\times \Si$ a $Q$, cioè: $\pt{q,a}\mapsto p$ (solo in un certo stato leggo un simbolo);
    \item $q_0 \in Q$ è lo stato iniziale;
    \item $F \subseteq Q$ è un insieme di stati \textbf{finali} (stati di accettazione);
\end{itemize}
\begin{critical}
    Per ogni stato deve essere presente una e una sola transizione uscente etichettata per ogni lettera dell'alfabeto.
    \emptyln{}
    Nell'esempio \cref{ex:first-ex} lo stato $q_0$ non può avere due transizioni con etichetta $1$ e non può mancare la transizione con l'etichetta $1$.
\end{critical}
\begin{example}{Un automa A che accetta $L=\pg{x01y: x,y \in \pg{0,1}^*}$}{first-ex}
    L'automa $A = \pt{\pg{q_0,q_1,q_2},\pg{0,1},\dt,q_0,\pg{q_1}}$ definito tramite una \textbf{tabella di transizione}:
    \begin{ctable}{r||c|c}
        & 0 & 1 \\
        \midrule
        \midrule
        $\rightarrow q_0$ & $q_2$ & $q_0$ \\
        $\star q_1$       & $q_1$ & $q_1$ \\
        $q_2$             & $q_2$ & $q_1$ \\
    \end{ctable}
    Lo stesso automa $A$ può essere definito tramite un \textbf{diagramma di transizione}:
    \begin{cimage}
        \includegraphics[width=.8\textwidth]{assets/dfa-ex-diag.png}
    \end{cimage}
\end{example}
Un automa a stati finiti accetta una stringa $w = a_1a_2\cdots a_n$ se esiste un cammino nel diagramma di transizione che:
\begin{enumerate}
    \item inizia nello \textbf{stato iniziale};
    \item finisce in uno \textbf{stato finale};
    \item ha una \textbf{sequenza} di etichette $a_1a_2\cdots a_n$;
\end{enumerate}

\subsubsection{Funzione di Transizione Estesa}\label{sub:dfa_funzione_di_transizione_estesa} % (fold)
La \textbf{funzione d transizione} $\dt$ può essere estesa a $\hat\dt$ che opera su stati e stringhe, invece che su stati e simboli.
\begin{description}
    \item[Base] $\hat\dt\pt{q,\eps} = q$
    \item[Induzione] $\hat\dt\pt{q,xa} = \dt\pt{\hat\dt\pt{q,x}a}$
\end{description}
Formalmente \hyperref[def:ling-reg]{il linguaggio accettato da un automa} a stati finiti deterministico $A$ è:
\[
    L\pt A = \pg{w : \hat\dt\pt{q_0,w} \in F}
\]
% subsubsection Funzione di Transizione Estesa (end)
%
\begin{definition}{Linguaggi Regolari}{ling-reg}
    Tutti i linguaggi accettati da automi a stati finiti sono detti \textbf{linguaggi regolari}
\end{definition}
\begin{example}{Funzione di Transizione Induttiva}{}
    \begin{cimage}
        \includegraphics[width=.8\textwidth]{assets/dfa-ex-diag.png}
    \end{cimage}
    \begin{description}
        \item[Base] $\hat\dt\pt{q,\eps} = q$
        \item[Induzione] $\hat\dt\pt{q,xa} = \dt\pt{\hat\dt\pt{q,x}a}$
    \end{description}
    Calcoliamo $\hat\dt\pt{q_0,0110}$
    \[
        \begin{aligned}
            &\hat\dt\pt{q_0,\eps} = q_0 \\
            &\hat\dt\pt{q_0, \red{\eps}0} = \dt\pt{\underbrace{\hat\dt\pt{q_0,\eps}}_{\red{q_0}}, 0} = \dt\pt{q_0, 0} = \red{q_2} \\
            &\hat\dt\pt{q_0, \red{0}1} = \dt\pt{\underbrace{\hat\dt\pt{q_0,0}}_{\red{q_2}}, 1} = \dt\pt{q_2, 1} = \red{q_1} \\
            &\hat\dt\pt{q_0, \red{01}1} = \dt\pt{\underbrace{\hat\dt\pt{q_0,01}}_{\red{q_1}}, 1} = \dt\pt{q_1, 1} = \red{q_1} \\
            &\hat\dt\pt{q_0, \red{011}0} = \dt\pt{\underbrace{\hat\dt\pt{q_0,011}}_{\red{q_1}}, 0} = \dt\pt{q_1, 0} = \red{q_1} \\
        \end{aligned}
    \]
\end{example}

\subsection{Automi a Stati Finiti Non Deterministici}\label{sub:lr_automi_a_stati_finiti_non_deterministici} % (fold)
Un \textbf{\ac{nfa}} accetta una stringa se, tra i tanti possibili, esiste un cammino che conduce a uno stato finale.
\begin{example}{}{nfa}
    Un automa accetta tutte e solo le stringhe che finiscono con \texttt{01}.
    \begin{cimage}
        \includegraphics[width=.8\textwidth]{assets/20260924_132157.png}
    \end{cimage}
    Ecco cosa succede quando l'automa elabora l'input \texttt{00101}:
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_132315.png}
    \end{cimage}
\end{example}
\begin{definition}{NFA}{}
    Formalmente, un \ac{nfa} è una quintupla
    \[
        A = \pt{Q,\Si,\dt,q_0,F}
    \]
    dove:
    \begin{itemize}
        \item $Q$ è un insieme finito di stati;
        \item $\Si$ è un \hyperref[def:alfabeto]{alfabeto} finito;
        \item $\dt$ è una \textbf{funzione di transizione} da $Q\times\Si$ all'insieme dei sottoinsiemi di $Q$, cioè: $\pt{q,a} \mapsto Q'$ con $Q' \subseteq Q$;
        \item $q_0 \in Q$ è lo stato \textit{iniziale};
        \item $F \subseteq Q$ è un insieme di stati \textit{finali};
    \end{itemize}
\end{definition}
\begin{example}{L'\hyperref[ex:nfa]{NFA precedente}}{}
   \[
       \pt{\pg{q_0,q_1,q_2}, \pg{0,1}, \dt, q_0, \pg{q_2}}
   \]
   dove $\dt$ è la funzione di transizione:
   \begin{ctable}{r||c|c}
       & 0 & 1 \\
       \midrule
       \midrule
       $\to q_0$ & $\pg{q_0,q_1}$ & $\pg{q_0}$ \\
       $q_1$ & $\emptyset$ & $\pg{q_2}$ \\
       $\star q_2$ & $\emptyset$ & $\emptyset$ \\
   \end{ctable}
\end{example}
\begin{note}[Lo stato Pozzo]
    Stato che durante un'esecuzione di un \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA}, non ha transizioni che cambiano stato.
    \begin{critical}
        All'interno dei \ac{nfa} lo stato pozzo viene rappresentato dall'assenza di transizioni
    \end{critical}
\end{note}
\subsubsection{Funzione di Transizione Estesa}\label{sub:nfa_funzione_di_transizione_estesa} % (fold)
La \textbf{funzione di transizione estesa} $\hat\dt$ opera su stati e stringhe (invece di stati e simboli).
\begin{description}
    \item[Base:] $\hat\dt\pt{q,\eps} = \pg{q}$
    \item[Induzione:] $\hat\dt\pt{q,xa} = \displaystyle\bigcup_{p\in\hat\dt\pt{q,x}}{\dt\pt{p,a}}$
\end{description}
Più formalmente, il \textbf{linguaggio accettato} da un automa a stati finiti non deterministico $A$ è
\[
    L\pt A = \pg{w: \hat\dt\pt{q_0,w} \cap F \neq \emptyset}
\]
\begin{example}{Calcoliamo $\hat\dt\pt{q_0,0010}$}{}
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_132157.png}
    \end{cimage}
    \begin{description}
        \item[Base:] $\hat\dt\pt{q,\eps} = \pg{q}$
        \item[Induzione:] $\hat\dt\pt{q,xa} = \displaystyle\bigcup_{p\in\hat\dt\pt{q,x}}{\dt\pt{p,a}}$
    \end{description}
    \[
        \begin{aligned}
            &\hdt\pt{q_0,\eps} = q_0 \\
            &\hdt\pt{q_0,\eps0} = \dt\pt{q_0,0} = q_0,q_1 \\ 
            &\hdt\pt{q_0,\eps00} = \dt\pt{q_0,0} \cup \dt\pt{{q_0,0}} = q_0,q_1 \cup \emptyset = q_0,q_1 \\
            &\hdt\pt{q_0,\eps001} = \dt\pt{q_0,1} \cup \dt\pt{q_1,1} = q_0 \cup q_2 = q_0,q_2 \\
            &\hdt\pt{q_0,\eps0010} = \dt\pt{q_0,0} \cup \dt\pt{q_2,0} = q_0,q_1 \cup \emptyset = q_0,q_1
        \end{aligned}
    \]
\end{example}

\subsection{Equivalenza di DFA e NFA}\label{sub:lr_equivalenza_di_dfa_e_nfa} % (fold)
Sorprendentemente, per ogni \ac{nfa} $N$ c'è un \ac{dfa} $D$, tale che $L\pt D = L\pt N$, e viceversa: questo comportamento comporta una \textit{costruzione a sottoinsiemi}.
\begin{theorem}{}{}
    Un automa $A$ può essere costruito da un altro automa $A$, dato un \hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA}
    \[
        N = \pt{Q_N,\Si,\dt_N,q_0,F_N}
    \]
    costruiremo un \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA}
    \[
        D = \pt{Q_D,\Si,\dt_D,\pg{q_0},F_D}
    \]
    tali che
    \[
        L\pt D = L\pt N
    \]
    con:
    \begin{itemize}
        \item $Q_D = \pg{S:S\subseteq Q_N}$
        \item $F_D = \pg{S \subseteq Q_N : S \cap F_N \neq \emptyset}$
        \item Per ogni $S \subseteq Q_N$ e $a \in \Si$:
            \[
                \dt_D\pt{S,a} = \displaystyle\bigcup_{p\in S}\dt_N\pt{p,a}
            \]
    \end{itemize}
\end{theorem}
\begin{example}{Costruire un \ac{dfa} da un \ac{nfa}}{dfa-to-fna}
    Costruiamo $\dt_D$ e (e $F_D$) dall'\hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA}:
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_132157.png}
    \end{cimage}
    Costruiamo $\dt_D$ dall'\hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA} già visto:
    \begin{ctable}{r||c|c}
        & 0 & 1 \\
        \midrule
        \midrule
        $\emptyset$ & $\emptyset$ & $\emptyset$ \\
        $\to\pg{q_0}$ & $\pg{q_0,q_1}$ & $\pg{q_0}$ \\
        $\pg{q_1}$ & $\emptyset$ & $\pg{q_2}$ \\
        $\star\pg{q_2}$ & $\emptyset$ & $\emptyset$ \\
        $\pg{q_0, q_1}$ & $\pg{q_0,q_1}$ & $\pg{q_0, q_2}$ \\
        $\star\pg{q_0, q_2}$ & $\pg{q_0,q_1}$ & $\pg{q_0}$ \\
        $\star\pg{q_1, q_2}$ & $\emptyset$ & $\pg{q_2}$ \\
        $\star\pg{q_0,q_1, q_2}$ & $\pg{q_0,q_1}$ & $\pg{q_0,q_2}$ \\
    \end{ctable}
    \begin{note}
        Gli stati di $D$ corrispondono a sottoinsiemi di stati $N$, ma potevamo denotare gli stati di $D$ in un altro modo (per esempio $A-H$):
        \begin{ctable}{r||c|c}
            & 0 & 1 \\
            \midrule
            \midrule
            $A$ & $A$ & $A$ \\
            $\star B$ & $E$ & $B$ \\
            $C$ & $A$ & $D$ \\
            $\star D$ & $A$ & $A$ \\
            $E$ & $E$ & $F$ \\
            $\star F$ & $E$ & $B$ \\
            $\star G$ & $A$ & $D$ \\
            $\star H$ & $E$ & $F$ \\
        \end{ctable}
    \end{note}
    Possiamo spesso evitare la crescita esponenziale degli stati costruendo la tabella di transizione per $D$ solo per stati accessibili $S$ come segue:
    \begin{description}
        \item[Base:] $S = \pg{q_0}$ è accessibile in $D$
        \item[Induzione:] Se lo stato $S$ è accessibile, lo sono anche gli stati $\dt_D\pt{S,a}$ per ogni $a \in \Si$
    \end{description}
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_170516.png}
        \caption{Sottoinsieme \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} solamente con stati accessibili}\label{fig:20260924_170516}
    \end{cimage}
\end{example}
\begin{theorem}{}{}
    Sia $D$ il \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} ottenuto da un \hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA} $N$ con la costruzione a sottoinsiemi, allora:
    \[
        L\pt D = L \pt N
    \]
\end{theorem}
\begin{theorem}{}{}
    Un \hyperref[def:linguaggio]{linguaggio} è accettato da un \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} se e solo se $L$ è accettato da un \hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA}
\end{theorem}
\begin{note}
    Il numero di stati del \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} equivalente a un \hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA} con $n$ stati è, nel caso peggiore, pari a $2^n$ stati
\end{note}

\subsection{FA con Transizioni Epsilon}\label{sub:lr_fa_con_transizioni_epsilon} % (fold)
\begin{definition}{Transizione Epsilon}{}
    Una \textbf{transizione epsilon} permette alla transizione di non consumare caratteri durante l'esecuzione
\end{definition}
Un $\eps$-\hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA} che accetta numeri decimali consiste di:
\begin{enumerate}[ref=(\arabic*)]
    \item un segno $+$ o $-$, opzionale;
    \item\label{tra-cifre} una stringa di cifre decimali;
    \item un punto decimale;
    \item\label{tra-stringa} un'altra stringa di cifre decimali;
\end{enumerate}
\begin{note}
    Una delle stringhe \cref{tra-cifre} o \cref{tra-stringa} è opzionale
\end{note}
\begin{cimage}
    \includegraphics[width=.6\textwidth]{assets/20260924_175200.png}
\end{cimage}
\begin{definition}{$\eps$-NFA}{}
    Un $\eps$-NFA è una quintupla $\pt{Q,\Si,\dt,q_0,FF}$ dove $\dt$ è una funzione da $Q \times \pt{\Si \cup \pg{\eps}}$ all'insieme dei sottoinsiemi di $Q$
\end{definition}
\begin{example}{}{}
    L'$\eps$-NFA \hyperref[ex:dfa-to-fna]{della pagina precedente} 
    \[
        E = \pt{\pg{\vc{q_0}{q_5}}, \pg{.,+,-,0,1,\dots,9},\dt,q_0,\pg{q_5}}
    \]
    dove la tabella delle transizioni per $\dt$ è:
    \begin{ctable}{r||c|c|c|c}
        & $\eps$ & $+,-$ & $.$ & $\vc{0}{9}$ \\
        \midrule
        \midrule
        $\to q_0$ & $\pg{q_1}$ & $\pg{q_1}$ & $\emptyset$ & $\emptyset$ \\
        $q_1$ & $\emptyset$ & $\emptyset$ & $\pg{q_2}$ & $\pg{q_1, q_4}$ \\
        $q_2$ & $\emptyset$ & $\emptyset$ & $\emptyset$ & $\pg{q_3}$ \\
        $q_3$ & $\pg{q_5}$ & $\emptyset$ & $\emptyset$ & $\pg{q_3}$ \\
        $q_4$ & $\emptyset$ & $\emptyset$ & $\pg{q_3}$ & $\emptyset$ \\
        $\star q_5$ & $\emptyset$ & $\emptyset$ & $\emptyset$ & $\emptyset$ \\
    \end{ctable}
\end{example}

\subsubsection{Epsilon-Chiusura}\label{sub:lr_epsilon_chiusura} % (fold)
L'operatore \textbf{epsilon-chiusura} chiude uno stato aggiungendo tutti gli stati raggiungibili da lui tramite una sequenza di $\eps\eps\dots\eps$.
\begin{definition}{$ECLOSE(q)$ Induttiva}{}
    \begin{description}
        \item[Base:] $q \in ECLOSE\pt q$
        \item[Induzione:] $q \in ECLOSE\pt q$ e $r \in \dt\pt{p,\eps} \Rightarrow r \in ECLOSE\pt q$
    \end{description}
\end{definition}
\begin{example}{$\eps$-chiusura}{}
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_181036.png}
    \end{cimage}
    \[
        ECLOSE(1) = \pg{1,2,3,4,6}
    \]
\end{example}
\begin{definition}{$\dt$ per $\eps$-NFA induttiva}{}
    \begin{description}
        \item[Base:] $\hdt\pt{q,\eps} = ECLOSE(q)$
        \item[Induzione:] $\hdt\pt{q,xa} = \displaystyle\bigcup_{p\in\hdt\pt{q,x}}\pt{\bigcup_{t \in\dt\pt{p,a}}ECLOSE(t)}$
    \end{description}
    Il linguaggio $L$ accettato è ancora definito come:
    \[
        \pg{w:\hdt\pt{q_0,w} \cap F \neq \emptyset}
    \]
\end{definition}
\begin{example}{}{}
    Il seguente $\eps$-NFA $E$
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_183850.png}
    \end{cimage}
    corrisponde al \ac{dfa} $D$
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20260924_184012.png}
    \end{cimage}
\end{example}
\begin{theorem}{}{}
    Un \hyperref[def:linguaggio]{linguaggio} $L$ è accettato da un \hyperref[sub:lr_fa_con_transizioni_epsilon]{$\eps$-NFA} $E$ se e solo se $L$ è accettato da un \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA}. 
\end{theorem}

\section{Espressioni Regolari}\label{sec:lr_espressioni_regolari} % (fold)
Un FA (\hyperref[sub:lr_automi_a_stati_finiti_non_deterministici]{NFA} o \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA}) è un metodo per costruire una macchina che riconosce \hyperref[def:ling-reg]{linguaggi regolari}.
\begin{definition}{Espressioni Regolari}{reg-exp}
    Un \textbf{espressione regolare} offre un modo dichiarativo di esprimere un \hyperref[def:ling-reg]{linguaggio regolare}, ovvero le \red{stringhe da accettare}.
\end{definition}
\begin{example}{}{}
    \[
        01* + 10*
    \]
    Le espressioni regolari sono usate in:
    \begin{itemize}
        \item comandi di ricerca, come \texttt{grep} nei sistemi \textit{UNIX} o nei browser di ricerca;
        \item strumenti per l'analisi lessicale di linguaggi, come \ac{lex}, \ac{flex} e \ac{antlr};
    \end{itemize}
\end{example}
\begin{description}
    \item[Unione] Si indica con $L \cup M$ e rappresenta l'insieme delle stringhe che sono in $L$ oppure in $M$.
        \[
            L \cup M = \pg{w:w\in L \text{ o } w \in L}
        \]
    \item[Concatenazione] È l'insieme delle stringhe che si possono formare prendendo una qualunque stringa in $L$ e concatenandola con qualsiasi stringa in $M$.
        \[
            LM = \pg{w:w = xy,\, x\in L,\,y \in M}
        \]
        \begin{example}{}{}
            \[
                \underbrace{\pg{ab, c}}_{L}\cdot \underbrace{\pg{d, ef}}_{M} = \underbrace{\pg{abd,\,abef,\,cd,\,cef}}_{L\cdot M}
            \]
        \end{example}
    \item[Potenze] 
        \[
            L^0 = \pg{\eps},\,L^3 = L,L^{\inc k} = L \cdot L^k
        \]
        \begin{example}{}{}
            \[
                \pg{ab,\,c}^2 = \pg{ab,\,c} \cdot \pg{ab,\,c} = \pg{abab,\,abc,\,cab,\,cc}
            \]
        \end{example}
    \item[Chiusura di Kleene (star)] $L^*$ rappresenta l'insieme delle stringhe che possono essere formate prendendo un numero qualsiasi da $L$, eventualmente con ripetizioni, e concatenandole tutte.
        \[
            L^* = L^0 \cup L^1 \cup L^2 \cup \cdots \cup L^n = \displaystyle\bigcup_{i\ge0}{L^i}
        \]
        \begin{example}{}{}
            \[
                \pg{ab,\,c}^* = \pg{\underbrace{\eps}_{L^0},\,\underbrace{ab,\,c}_{L^1},\,\underbrace{abab,\,abc,\,cab,\,cc}_{L^2},\,\underbrace{ababab,\,\dots}_{L^3}}
            \]
        \end{example}
        \begin{critical}
            \[
                \emptyset^* = \displaystyle\bigcup_{i\ge0}{\emptyset^i} = \pg{\eps}
            \]
            con
            \[
                \emptyset^i = \emptyset \qquad \text{per ogni } i > 0
            \]
        \end{critical}
\end{description}

\begin{multicols}{2}
    \begin{itemize}
        \item $L\cup M = M \cup L$ \textit{commutativa};
        \item $\pt{L \cup M} \cup N = L \cup \pt{M \cup N}$ \textit{associativa};
    \end{itemize}
\end{multicols}
\begin{itemize}
    \item $\pt{LM}N = L\pt{MN}$ la concatenazione è \textit{associativa};
        \begin{note}[La concatenazione non è commutativa]
            Esistono $L$ e $M$ tali che $LM \ne ML$
        \end{note}
\end{itemize}
\begin{multicols}{2}
    \begin{itemize}
        \item $\emptyset \cup L = L \cup \emptyset = L$ $\emptyset$ è l'\textit{identità} per l'\red{unione};
        \item $\pg \eps L = L \pg\eps = L$ $\pg\eps$ è l'\textit{identità} sinistra e destra per la \red{concatenazione};
        \item $\emptyset L = L\emptyset = \emptyset$ $\emptyset$ è l'\textit{annichilatore} sinistro e destro per la \red{concatenazione};
        \item $L \pt{M \cup N} = LM \cup LN$, la \textit{concatenazione} è distributiva a sinistra sull'unione;
        \item $\pt{M\cup N}L = ML \cup NL$, la \textit{concatenazione} è distributiva a destra sull'unione;
        \item $L\cup L = L$, l'\textit{unione} è idempotente;
        \item $\emptyset^* = \pg \eps,\,\pg\eps^* = \pg\eps$
        \item $L^+ = LL^* = L^*L,\,L^* = L^+ \cup\pg\eps$
        \item $\pt{L^*}^* = L^*$
    \end{itemize}
\end{multicols}

\subsection{Costruire le Espressioni Regolari}\label{sub:lr_costruire_le_espressioni_regolari} % (fold)
L'algebra delle espressioni regolari usa costanti e variabili che indicano linguaggi e operatori per le operazioni di unione, punto e star.
\\
Definizione in maniera ricorsiva delle espressioni regolari:
\begin{description}
    \item[Base] Consiste di 3 parti:
        \begin{itemize}
            \item le costanti $\eps$ e $\emptyset$ sono espressioni regolari, indicanti i linguaggi $\pg\eps$ e $\emptyset$
                \[
                    L\pt\eps = \pg\eps \qquad L\pt\emptyset = \emptyset
                \]
            \item se $a$ è un simbolo qualsiasi, allora $a$ è un espressione regolare.
                Tale espressione denota il linguaggio $\pg a$, cioè:
                \[
                    L\pt{\mathbf{a}} = \pg a
                \]
                \begin{note}
                    Si usa il grassetto per indicare un'espressione corrispondente a un simbolo
                \end{note}
            \item una variabile, come $L$, rappresentante un linguaggio arbitrario
        \end{itemize}
    \item[Induzione] Il passo induttivo è formato da 4 parti, una per ognuno dei \red{tre operatori} e una per l'\red{introduzione delle parentesi}:
        \begin{itemize}
            \item se $E$ è un espressione regolare, allora $\pt E$ è un espressione regolare
                \[
                    L\pt{\pt E} = L\pt E
                \]
            \item se $E$ e $F$ sono espressioni regolari, allora $E+F$ è un espressione regolare
                \[
                    L\pt{E+F} = L\pt E \cup L\pt F
                \]
            \item se $E$ e $F$ sono espressioni regolari, allora $EF$ è un espressione regolare
                \[
                    L\pt{EF} = L\pt E \cdot L\pt F
                \]
                \begin{note}
                    Il punto $\cdot$ è facoltativo in quanto nelle espressioni regolari $0\cdot1 = 01$ e rappresenta il linguaggio $\pg{01}$
                \end{note}
            \item Se $E$ è un espressione regolare, allora $E^*$ è un espressione regolare
                \[
                    L\pt{E^*} = \pt{L\pt E}^*
                \]
        \end{itemize}
\end{description}
\begin{example}{Costruire un espressione regolare}{}
    \[
        \begin{aligned}
            &E = \pt{0+1}\cdot \pt{1^*} \\
            &L\pt{0} = \pg{0} \\
            &L\pt{1} = \pg{1} \\
            &L\pt{1^*} = L\pt 1^* = \pg{1^*} = \bigcup_{i\ge0}{\pg{1}^i} = \pg{1^n | n\ge0}\,\,(1, 11, 111, ...) \\
            &L\pt{(0+1)} = L\pt{0+1} = L\pt 0 \cup L\pt 1 = \pg 0 \cup \pg 1 = \pg{0,1} \\
            &L\pt E = L\pt{(0+1)} \cdot L\pt{\pt{1^*}} = \pg{0,1} \cdot \pg{1^n | n \ge 0} = \pg{x1^n | x\in\pg{0,1} \text{con } n\ge 0}
        \end{aligned}
    \]
\end{example}
\begin{example}{Equivalenza di espressioni regolari}{}
    Dato il seguente linguaggio:
    \[
        L = \pg{w \in \pg{0,1}^*: 0 \text{ e } 1 \text{ alternati in } w}
    \]
    \[
        (01)^* + (10)^* + 0(10)^* + 1(01)^*
    \]
    In questa espressione vengono uniti 4 linguaggi differenti:
    \begin{multicols}{2}
        \begin{itemize}
            \item $(01)^*$ comprende tutte le stringhe $01$ ripetute 0 o più volte;
            \item $(10)^*$ comprende tutte le stringhe $10$ ripetute 0 o più volte;
            \item $0(10)^*$ comprende tutte le stringhe che iniziano per 0, seguito da una (possibile) serie di $10$ (in pratica tutte le stringhe che iniziano e finiscono per 0);
            \item $1(01)^*$ comprende tutte le stringhe che iniziano per 1, seguito da una (possibile) serie di $01$ (in pratica tutte le stringhe che iniziano e finiscono per 1);
        \end{itemize}
    \end{multicols}
    \begin{note}
        Grazie alla ``somma'' è possibile effettuare l'unione di tutti i linguaggi
    \end{note}
    Esiste un modo alternativo, che produce un'\hyperref[def:reg-exp]{espressione regolare} diversa e più concisa.
    Effettuando una concatenazione a sinistra con l'espressione $\eps + 1$ possiamo aggiungere un $1$ facoltativo all'inizio.
    Analogamente aggiungendo uno $0$ facoltativo alla fine, con l'espressione $\eps + 0$ è possibile ottenere il linguaggio sopraelencato.
    \begin{note}
        Ricorrendo alla definizione dell'operatore di somma $(+)$ (unione) si ottiene:
        \[
            L\pt{\inc\eps} = L\pt\eps\cup L\pt 1 = \pg\eps\cup\pg 1 = \pg{\eps,1}
        \]
        Se si concatena questo linguaggio con qualsiasi altro linguaggio $L$, la scelta $\eps$ dà tutte le stringhe in $L$, mentre la scelta $1$ dà $1w$ per ogni stringa $w$ in $L$
    \end{note}
    Detto ciò un'altra espressione per l'insieme di stringhe che alternano 0 e 1 è la seguente:
    \[
        \pt{\inc\eps}\pt{01}^*\pt{\eps + 0}
    \]
    \begin{warning}
        Per garantire che gli operatori siano raggruppati opportunamente, sono necessarie le parentesi prima e dopo ogni espressione aggiunta
    \end{warning}
\end{example}

\subsubsection{Precedenza degli Operatori}\label{sub:lr_precedenza_degli_operatori} % (fold)
Anche l'\hyperref[sub:lr_leggi_algebriche_per_i_linguaggi]{algebra} delle \hyperref[def:reg-exp]{espressioni regolari} hanno un ordine di precedenza, come nelle altre algebre, il che significa che gli operatori sono associati ai loro operandi in un ordine particolare.
\begin{example}{}{}
    Sappiamo che $xy+z$ raggruppa il prodotto $xy$ prima della somma, dunque è equivalente all'espressione $(xy) + z$ e \textbf{non} all'espressione $x(y + z)$.
\end{example}
\begin{warning}[In aritmentica si raggruppa partendo da sinistra $x-y-z = (x-y)-z$]
\end{warning}
L'\textbf{ordine di precedenza} degli operatori è il seguente:
\begin{enumerate}
    \item l'operatore \textbf{star} ($^*$) o \textbf{chiusura di Kleene} ha il livello più alto di proprietà;
    \item la concatenazione \textbf{punto} ($. \text{ o } \cdot$): dopo aver raggruppato tutti gli \textbf{star}, si effettua la concatenazione in qualsiasi ordine, essendo un'operazione \textit{associativa} non importa l'ordine di esecuzione (in caso di scelta è preferibile sempre da \red{sinistra});
    \item l'ultimo operatore è l'\textbf{unione} ($+ \text{ o } \cup$: dato che anche l'unione è \textit{associativa}, non è importante in quale ordine vengono raggruppate le unioni consecutive (in ogni caso si assume che il raggruppamento è avvenuto partendo da sinistra);
\end{enumerate}

\subsubsection{Da DFA a Espressioni Regolari}\label{sub:lr_da_dfa_a_espressioni_regolari} % (fold)

\subsection{Chiusura dei Linguaggi Regolari}\label{sub:lr_chiusura_dei_linguaggi_regolari} % (fold)
\begin{theorem}{Chiusura rispetto all'unione}{lr-union}
    Se $L$ e $M$ sono linguaggi regolari, allora è regolare anche $L\in M$
    \begin{proof}
        Poiché $L$ e $M$ sono regolari, essi sono descritti da espressioni regolari, poniamo $L = L(R)$ e $M = L(S)$.
        Allora $L\in M = L(R+S)$ per la definizione dell'operatore $\mathbf{+}$ delle \hyperref[def:reg-exp]{espressioni regolari}.
    \end{proof}
\end{theorem}
\begin{theorem}{Chiusura rispetto al complemento}{lr-complem}
    Se $L$ è un linguaggio regolare sull'alfabeto $\Si$, allora anche $\overline L = \Si[*] - L$ è regolare
    \begin{proof}
        Sia $L = L(A)$ per un \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} $A = \pt{Q,\Si,\dt,q_0,F}$.
        Allora $\overline L = L(B)$, dove $B$ è il \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} $\pt{Q,\Si,\dt,q_0,Q-F}$.
        In altre parole $B$ è esattamente come $A$, ma gli stati accettati di $A$ sono diventati stati non accettanti di $B$, e viceversa.
        Allora $w$ è in $L(B)$ se e solo se $\hat\dt\pt{q_0,w}$ è in $Q-F$, il che succede se e solo se $w$ non è in $L(A)$.
    \end{proof}
    \begin{warning}
        È importante che $\hat\dt\pt{q_0,w}$ sia sempre definito; vale a dire che non ci siano transizioni mancanti in $A$
        \begin{critical}
            Nel caso in cui manchino delle transizioni, allora alcune stringhe potrebbero non condurre ad alcuno stato di $A$, né accettante né non accettante, e le stringhe verrebbero a mancare sia in $L(A)$ sia in $L(B)$
        \end{critical}
    \end{warning}
\end{theorem}

\begin{theorem}{Chiusura rispetto all'intersezione}{lr-inters}
    Se $L$ e $M$ sono linguaggi regolari, allora anche $L \cap M$
    \begin{proof}[DImostrazione Banale]
        Per la \textit{legge di DeMorgan}, si può dire che:
        \[
            L\cap M = \overline{\overline L \cup \overline M}
        \]
    \end{proof}
    \begin{proof}
        Siano $L$ e $M$ i linguaggi degli automi \\$A_L = \pt{Q_L,\Si,\dt_L,q_L,F_L}$ e $A_M = \pt{Q_M,\Si,\dt_M,q_M,F_M}$.
        Inoltre per semplicità si assume che entrambi siano \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA}.
        \emptyln{}
        Per $L\cap M$ costruiremo un automa $A$ che simuli sia $A_L$ sia$A_M$.
        Gli stati di $A$ sono coppie di stati, il primo di $A_L$ e il secondo di $A_M$.
        Per stabilire le transizioni di $A$, si suppone che sia nello stato $\pt{p,q}$, dove $P$ è lo stato di $A_L$ e $q$ è lo stato di $A_M$.
        \emptyln{}
        Se $a$ è il simbolo dell'input, vediamo come si comporta $A_L$ a fronte dell'input $a$; poniamo che vada nello stato $s$.
        Vediamo anche come si comporta $A_M$ a fronte dell'input $a$; poniamo che compia una transizione nello stato $t$.
        Allora il prossimo stato di $A$ sarà $\pt{s,t}$.
        In questo modo $A$ ha simulato l'effetto sia di $A_L$ sia di $A_M$.
        \begin{cimage}
            \includegraphics[width=.4\textwidth]{assets/20261008_105953.png}
            \caption{}\label{fig:20261008_105953}
        \end{cimage}
        Poiché vogliamo che l'automa in \cref{fig:20261008_105953} accetti se e solo se entrambi gli automi accettano, selezioniamo come stati accettanti tutte quelle coppie $\pt{p,q}$ tali che $p$ sia uno stato accettante di $A_L$ e $q$ sia uno stato accettante di $A_M$:
        \[
            A = \pt{Q_L\times Q_M,\Si,\dt,\pt{q_L,q_M},F_L\times F_M}
        \]
        dove
        \[
            \dt_{L\cap M}\pt{\pt{p,q},a} = \pt{\dt_L\pt{p,a},\dt_M\pt{q,a}}
        \]
    \end{proof}
\end{theorem}

\begin{theorem}{Chiusura rispetto alla differenza}{lr-diff}
    Se $L$ e $M$ sono linguaggi regolari, allora anche $L-M$ è regolare
    \begin{proof}
        Osserviamo che $L-M = L \setminus M = L\cap \overline M$.
        Per il teorema \hyperref[th:lr-complem]{del complemento} $\overline M$ è regolare, e per il teorema \hyperref[th:lr-inters]{dell'intersezione} $L\cap \overline M$ è regolare.
        Di conseguenza $L-M$ è regolare.
    \end{proof}
\end{theorem}

\subsection{Inversione}\label{sub:lr_inversione} % (fold)
\begin{definition}{Inversione}{lr-inversione}
    L'\textbf{inversione} di una stringa $a_1a_2\dots a_n$ è la stringa \red{scritta al contrario}, ossia
    \[
        a_na_{\dec n}\dots a_1
    \]
    \begin{note}[Si denota con $w^R$ l'inversione della stringa $w$]
    \end{note}
\end{definition}
\begin{theorem}{}{}
    Se $L$ è un linguaggio regolare, lo è anche $L^R$
    \begin{proof}
        Sia il linguaggio $L$ riconosciuto da un FA $A$.
        Modifichiamo $A$ per renderlo un FA per $L^R$:
        \begin{enumerate}
            \item giriamo tutti gli archi;
            \item rendiamo il vecchio stato iniziale l'unico stato finale;
            \item creiamo un nuovo stato iniziale $p_0$ con $\dt(p_0,\eps) = F$ (i vecchi stati finali);
        \end{enumerate}
    \end{proof}
\end{theorem}

\end{enumerate}

\subsection{Verificare se un Linguaggio Regolare è vuoto}\label{sub:lr_verificare_se_un_linguaggio_regolare_vuoto} % (fold)
\begin{remark}{}{}
    \begin{description}
        \item[Base] Lo stato iniziale è sicuramente raggiungibile dallo stato iniziale
        \item[Induzione] Se lo stato $q$ è raggiungibile dallo stato iniziale e c'è un arco da $q$ a $p$ con etichetta arbitraria, anche $p$ è raggiungibile
    \end{description}
\end{remark}
In questo modo si determina l'insieme degli stati raggiungibili.
Se fra loro c'è uno stato accettante, la risposta è ``no'' (il linguaggio dell'automa \red{non è vuoto}), altrimenti è ``si''.
\emptyln{}
Oppure un altro modo è quello di osservare un'espressione regolare $E$ di lunghezza $n$.
\begin{remark}{}{}
    \begin{description}
        \item[Base] $\emptyset$ denota il linguaggio vuoto: $\eps$ e $a$ no, per qualsiasi simbolo di input di $a$
        \item[Induzione] Sia $E$ un'espressione regolare, si devono considerare quattro casi, che corrispondono ai modi in cui si può costruire $E$:
            \begin{enumerate}
                \item $E = F + G$ In questo caso $L(E)$ è vuoto se e solo se sia $L(F)$ che $L(G)$ sono vuoti;
                \item $E = FG$ In questo caso $L(E)$ è vuoto se e solo se $L(F)$ o $L(G)$ sono vuoti;
                \item $E = F^*$ In questo caso $L(E)$ \textbf{non è mai vuoto} perché $\eps \in L(E)$;
                \item $E = (F)$ In questo caso $L(E)$ è vuoto se e solo se $L(F)$ è vuoto, dato che sono lo stesso linguaggio;
            \end{enumerate}
    \end{description}
\end{remark}

\section{Equivalenza e Minimizzazione di Automi}\label{sec:lr_equivalenza_e_minimizzazione_di_automi} % (fold)
\subsection{Equivalenza degli Stati}\label{sub:lr_equivalenza_degli_stati} % (fold)
\begin{theorem}{}{stati-equiv}
    Diciamo che due stati $p$ e $q$ sono \textbf{equivalenti} se per ogni stringa di \textit{input} $w$, $\hat\dt\pt{p,w}$ è uno stato accettante se e solo se $\hat\dt\pt{q,w}$ è uno stato accettante.
\end{theorem}
\begin{theorem}{}{}
    Se due stati non sono equivalenti, diciamo che sono \textbf{distinguibili}.
    In altre parole lo stato $p$ è distinguibile da $q$ se esiste almeno una stringa $w$ tale che uno fra $\hat\dt\pt{p,w}$ e $\hat\dt\pt{q,w}$ è accettante e l'altro no.
\end{theorem}
L'algoritmo, chiamato \textit{algoritmo riempi-tabella}, scopre ricorsivamente le coppie di stati distinguibili di un \hyperref[sub:lr_automi_a_stati_finiti_deterministici]{DFA} $A = \pt{Q;\Si,\dt,q_0,F}$.
\begin{remark}{}{}
    \begin{description}
        \item[Base] Se $p$ è uno stato accettante e $q$ è non accettante, la coppia $\pg{p,q}$ è distinguibile.
        \item[Induzione] Siano $p$ e $q$ due stati tali che, per un simbolo di input $a$, $r = \dt\pt{p,a}$ e $s = \dt(q,a)$ sono stati che sappiamo essere distinguibili.
            Allora $\pg{p,q}$ è una coppia di stati distinguibili.
            \emptyln{}
            La regola vale perché dev'esserci una stringa $w$ che distingue $r$ da $s$; in altre parole uno solo tra $\hat\dt(r,w)$ e $\hat\dt(s,w)$ è accettante.
            Ma allora la stringa $aw$ distingue $p$ da $q$ perché $\hat\dt(p,aw)$ e $\hat\dt(q,aw)$ coincidono rispettivamente con $\hat\dt(r,w)$ e $\hat\dt(s,w)$.
    \end{description}
\end{remark}
\begin{example}{Che cosa si può dire su $A$ e $E$?}{}
    \begin{cimage}
        \includegraphics[width=.6\textwidth]{assets/20261009_133731.png}
    \end{cimage}
    \[
        \begin{aligned}
            &\hdt\pt{A,\eps} = A \notin F,\qquad\hdt\pt{E,\eps} = E \notin F \\
            &\hdt\pt{A,1} = F = \hdt\pt{E,1} \\
            &\text{Quindi: } \hdt\pt{A,1x} = \hdt\pt{E,1x} = \hdt\pt{F,x} \\
            &\hdt\pt{A,0} = B\notin F, \qquad \hdt\pt{E,0} = H \notin F \\
            &\hdt\pt{A,00} = G = \hdt\pt{E,00}, \qquad \hdt\pt{A,01} = C = \hdt\pt{E,01} \\
            &\text{Conclusione: } A \equiv E
        \end{aligned}
    \]
    \begin{remark}{}{}
        Considerando $A$ e $E$: non sono accettanti quindi $\eps$ non li distingue.
        A fronte di input $1$ entrambi vanno in $F$.
        Nessuna stringa che cominci per $1$ può quindi distinguerli perché, per ogni stringa $x$:
        \[
            \hdt\pt{A,1x} = \hdt\pt{E,1x}
        \]
        Per input che cominciano con $0$ $A$ va in $B$, $E$ in $H$.
        Nessuno dei due nuovi stati è accettante, perciò la stringa $0$ da sola non distingue $A$ da $E$.
        Anche $B$ e $H$ non sono d'aiuto: su input $1$ entrambi vanno in $C$ e su input $0$ entrambi vanno in $G$.
        Quindi nessun input che cominci per $0$ può distinguerli.
        \emptyln{}
        Concludiamo che nessuna stringa di input distingue $A$ da $E$ quindi i due stati sono \hyperref[th:stati-equiv]{equivalenti}.
    \end{remark}
\end{example}
