\chapter{Intro}\label{chap:intro} % (fold)
Providing solid knowledge and skills about \textbf{modelling}, \textbf{designing} and \textbf{programming concurrent systems}.
\begin{itemize}
    \item modelling and designing concurrent systems: software engineering view in particular;
    \item concurrent programming:
        \begin{itemize}
            \item multithreaded programming;
            \item asynchronous programming;
            \item message-passing and actor-based models;
        \end{itemize}
\end{itemize}
